cmake_minimum_required(VERSION 3.12)
project(TotalBattle2D)

set(CMAKE_CXX_STANDARD 20)
set(MIXER_PATH "${CMAKE_SOURCE_DIR}/libs/SDL3_mixer-3.2.0/x86_64-w64-mingw32")

add_executable(TotalBattle2D main.cpp
        Components.cpp
        Components.h
        State.cpp
        State.h
        Entity.cpp
        Entity.h
        Player.cpp
        Player.h
        TileMap.cpp
        TileMap.h
        FastNoiseLite.h
        Camera.cpp
        Camera.h
        Province.cpp
        Province.h
        Settlements.cpp
        Settlements.h
        Buildings.cpp
        Buildings.h
        UnitCardStats.cpp
        UnitCardStats.h
        TechnologyTree.cpp
        TechnologyTree.h
        Date.cpp
        Date.h
        WorldEvents.cpp
        WorldEvents.h
        Resources.cpp
        Resources.h
)

target_include_directories(TotalBattle2D PRIVATE "${MIXER_PATH}/include")

find_library(MIXER_LIBRARY NAMES SDL3_mixer PATHS "${MIXER_PATH}/lib" NO_DEFAULT_PATH)

target_link_libraries(TotalBattle2D PRIVATE
        SDL3
        SDL3_image
        SDL3_ttf
        sqlite3
        ${MIXER_LIBRARY}
)

set(ASSETS_SRC_DIR ${CMAKE_SOURCE_DIR}/assets)
set(ASSETS_BINARY_DIR ${CMAKE_BINARY_DIR}/assets)
add_custom_target(
        assets ALL
        COMMAND ${CMAKE_COMMAND} -E remove_directory ${ASSETS_BINARY_DIR}
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${ASSETS_SRC_DIR} ${ASSETS_BINARY_DIR}
        COMMENT "Clearing and copying game assets into the binary directory."
)
add_dependencies(${PROJECT_NAME} assets)

add_custom_command(TARGET TotalBattle2D POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${MIXER_PATH}/bin/SDL3_mixer.dll"
        $<TARGET_FILE_DIR:TotalBattle2D>
)
